cmake_minimum_required(VERSION 3.16)
project(DataServer LANGUAGES CXX)
configure_file(fields.json fields.json COPYONLY)
configure_file(field_groups.json field_groups.json COPYONLY)
find_package(Threads REQUIRED)
function(add_data_server target)
  add_executable(${target} src/main.cpp)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_link_libraries(${target} PRIVATE Threads::Threads)
  if(WIN32)
    target_link_libraries(${target} PRIVATE ws2_32 bcrypt)
  endif()
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endfunction()

add_data_server(data_server)
if(WIN32)
  add_data_server(data_server_lan)
  target_compile_definitions(data_server_lan PRIVATE DATA_SERVER_LAN)
  if(MINGW)
    target_link_options(data_server_lan PRIVATE -static)
  endif()
endif()
enable_testing()
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
  add_test(NAME protocol COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:data_server>)
endif()
